Data Warehouse Analytical SQL Queries

1. Dynamic League Table View (Window Functions)

CREATE VIEW vw_league_table AS
SELECT
    DENSE_RANK() OVER (
        ORDER BY s.points DESC, s.goal_difference DESC, s.goals_for DESC
    ) AS dynamic_rank,
    c.name AS club_name,
    c.short_name,
    s.played, s.won, s.drawn, s.lost,
    s.goals_for, s.goals_against, s.goal_difference,
    s.points, s.points_per_game, s.form
FROM standings s
JOIN clubs c ON s.club_id = c.club_id;

2. Golden Boot Top Scorers View

CREATE VIEW vw_top_scorers AS
SELECT
    ROW_NUMBER() OVER (
        ORDER BY p.goals DESC, p.assists DESC, p.appearances ASC
    ) AS scorer_rank,
    p.name AS player_name,
    c.name AS club_name,
    p.position, p.goals, p.assists,
    p.goal_contributions, p.goals_per_game
FROM players p
JOIN clubs c ON p.club_id = c.club_id
WHERE p.goals > 0;

3. Club Squad Performance Aggregation View

CREATE VIEW vw_club_analytics AS
SELECT
    c.name AS club_name,
    COUNT(DISTINCT p.player_id) AS squad_size,
    COALESCE(SUM(p.goals), 0) AS squad_goals,
    COALESCE(SUM(p.assists), 0) AS squad_assists,
    ROUND(AVG(p.appearances), 1) AS avg_player_appearances,
    c.capacity
FROM clubs c
LEFT JOIN players p ON c.club_id = p.club_id
GROUP BY c.club_id, c.name, c.capacity;
